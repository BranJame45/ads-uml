' -- Flujo inicial (Directo Actor -> Controlador) --
a_vendedor      --> ctrl            : 1. validarBtnRegistrarVenta(btnRegistrarVenta)
ctrl            --> b_sel_cliente   : 2. CI_VentanaSeleccionClienteShow()

' -- Rama Buscar Cliente (sin isset por R4) --
b_sel_cliente   --> ctrl            : 3a. [BUSCAR CLIENTE] buscarCliente(textoBusqueda)
ctrl            --> e_cliente       : 4a. buscarClientePorDocONombre(textoBusqueda)
ctrl            --> b_sel_cliente   : 5a. CI_VentanaSeleccionClienteShow(datosCliente)

' -- Rama Completar Reserva Existente (1º Cancelar, 2º ConfirmarPago) --
b_sel_cliente   --> ctrl            : 3b. [COMPLETAR RESERVA] Envía btnCompletarReserva
ctrl            --> e_venta         : 4b. obtenerReservasPendientes(id_cliente)
ctrl            --> e_detalle       : 5b. obtenerDetalleReserva(id_venta)
ctrl            --> b_cobro_reserva : 6b. CI_FormularioCobroReservaShow(datosReserva, detalleReserva)
a_vendedor      --> b_cobro_reserva : 7b-cancel. [alt-Cancelar] clickCancelar(btnCancelar) → vuelve a Selección
b_cobro_reserva --> ctrl            : 7b. [else-ConfirmarPago] ingresarDatos(metodo_pago, montoIngresado, btnConfirmarPago)
ctrl            --> e_venta         : 8b. registrarPago(id_venta, metodo_pago, monto)
ctrl            --> e_venta         : 9b. actualizarVentaCompletada(id_venta)
ctrl            --> e_producto      : 10b. liberarYDescontarStock(id_prod, cantidad)
ctrl            --> b_comprobante   : 11b. CI_VistaComprobanteShow(datosComprobante)
a_vendedor      --> b_comprobante   : 12b-1. [alt-Imprimir] clickImprimir(btnImprimir)
a_vendedor      --> b_comprobante   : 12b-2. [else-DescargarPDF] clickDescargarPDF(btnDescargarPDF)
a_vendedor      --> b_comprobante   : 12b-3. [else-Finalizar] clickFinalizar(btnFinalizar)
ctrl            --> b_sel_cliente   : 13b. CI_VentanaSeleccionClienteShow()

' -- Rama Iniciar Nueva Venta (1º Cancelar, 2º Aceptar) --
b_sel_cliente   --> ctrl            : 3c. [NUEVA VENTA] Envía btnIniciarNuevaVenta
ctrl            --> b_confirm       : 4c. [si sin cliente] CI_MensajeConfirmacionShow(Público General)
a_vendedor      --> b_confirm       : 5c-cancel. [alt-Cancelar] clickCancelar(btnCancelar) → vuelve a Selección
b_confirm       --> ctrl            : 5c. [else-Aceptar] Envía btnAceptar
ctrl            --> e_producto      : 6c. obtenerProductosDisponibles()
ctrl            --> b_catalogo      : 7c. CI_VentanaCatalogoCarritoShow(listadoProductos)

' -- Sub-ramas en Catálogo y Carrito --
b_catalogo      --> ctrl            : 8c-1. [BUSCAR ÍTEM] buscarItem(textoBusqueda)
ctrl            --> e_producto      : 9c-1. obtenerProductosFiltrados(textoBusqueda)
ctrl            --> b_catalogo      : 10c-1. CI_VentanaCatalogoCarritoShow(listadoFiltrado)

b_catalogo      --> ctrl            : 8c-2. [AGREGAR CARRITO] Envía btnAgregarCarrito(id_prod, cantidad)
ctrl            --> e_producto      : 9c-2. verificarStockProducto(id_prod, cantidad)
ctrl            --> b_catalogo      : 10c-2. CI_VentanaCatalogoCarritoShow(carritoActualizado)

b_catalogo      --> ctrl            : 8c-3. [MODIFICAR CANTIDAD] Envía btnModificarCantidad(id_prod, cantidad)
ctrl            --> e_producto      : 9c-3. verificarStockProducto(id_prod, cantidad)
ctrl            --> b_catalogo      : 10c-3. CI_VentanaCatalogoCarritoShow(carritoActualizado)

b_catalogo      --> ctrl            : 8c-4. [ELIMINAR ÍTEM] Envía btnEliminarItem(id_prod)
ctrl            --> b_catalogo      : 9c-4. CI_VentanaCatalogoCarritoShow(carritoActualizado)

b_catalogo      --> ctrl            : 8c-5. [PROCESAR VENTA] Envía btnProcesarVenta(tipoComprobante)
ctrl            --> b_cobro_final   : 9c-5. CI_FormularioCobroFinalShow(resumenVenta)
a_vendedor      --> b_cobro_final   : 10c-5-cancel. [alt-Cancelar] clickCancelar(btnCancelar) → limpia sesión y vuelve
b_cobro_final   --> ctrl            : 10c-5. [else-Confirmar] ingresarDatos(metodo_pago, montoIngresado, btnConfirmarTransaccion)
ctrl            --> e_venta         : 11c-5. [si Venta Total] registrarPago(id_venta, metodo_pago, monto)
ctrl            --> e_venta         : 12c-5. [si Venta Total] obtenerCorrelativoComprobante(tipo_comprobante)
ctrl            --> e_venta         : 13c-5. registrarVenta(id_cliente, id_usuario, num_comprobante, tipo_comprobante, total, adelanto, saldo, metodo_pago, estado)
ctrl            --> e_detalle       : 14c-5. registrarDetalleVenta(id_venta, id_prod, cantidad, precio_unitario, subtotal)
ctrl            --> e_producto      : 15c-5. separarStockProducto(id_prod, cantidad) / descontarStockFisico(id_prod, cantidad)
ctrl            --> b_sel_cliente   : 16c-5-res. [si Reserva] CI_VentanaSeleccionClienteShow()
ctrl            --> b_comprobante   : 16c-5. [si Venta Total] CI_VistaComprobanteShow(datosComprobante)
a_vendedor      --> b_comprobante   : 17c-5-1. [alt-Imprimir] clickImprimir(btnImprimir)
a_vendedor      --> b_comprobante   : 17c-5-2. [else-DescargarPDF] clickDescargarPDF(btnDescargarPDF)
a_vendedor      --> b_comprobante   : 17c-5-3. [else-Finalizar] clickFinalizar(btnFinalizar)
ctrl            --> b_sel_cliente   : 18c-5. CI_VentanaSeleccionClienteShow()

' -- Mensajes de sistema (éxito, aviso o validación fallida) --
ctrl            --> b_msg           : * CI_MensajeSistemaShow() en éxito, aviso o validación fallida
