  note right: SELECT id_prov, razon_social, ruc, telefono, correo FROM Proveedor WHERE estado = 'Activo' ORDER BY razon_social

    else (NO)
      :Consultar en la BD proveedores activos filtrados por razón social o RUC;
      note right: SELECT id_prov, razon_social, ruc, telefono, correo FROM Proveedor WHERE estado = 'Activo' AND (razon_social LIKE '%?%' OR ruc = ?)

      stop
    endif

  case (boton <Nuevo Proveedor>)
    :Validar btnNuevoProveedor;
    if (isset(btnNuevoProveedor)) then (False)
      :Mostrar panel <Error> "Acceso no permitido";
      stop
    else (TRUE)
      :Mostrar modal <Formulario de Proveedor> vacío con el campo RUC editable;
      if (Clic en ...) then (boton <Cancelar>)
        :Cerrar modal y mantener vista actual;
        stop

          if (¿Campos incompletos?) then (SI)
            :Mostrar panel <Error> "La razón social, RUC, teléfono, correo y dirección son obligatorios";

          else (NO)
            :Consultar en la BD si ya existe un proveedor con ese RUC;
            note right: SELECT id_prov FROM Proveedor WHERE ruc = ?

              note right: INSERT INTO Proveedor (razon_social, ruc, telefono, correo, direccion, estado) VALUES (?, ?, ?, ?, ?, 'Activo')

          endif
        endif
      endif
    endif
