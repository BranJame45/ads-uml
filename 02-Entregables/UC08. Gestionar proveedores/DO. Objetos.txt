' ============================================================
' RELACIONES NUMERADAS (orden del DS, priorizando camino negativo)
' ============================================================

' -- Flujo inicial (Directo Actor -> Controlador) --
a_encargado --> ctrl        : 1. validarBtnGestionarProveedores(btnGestionarProveedores)
ctrl        --> e_proveedor : 2. obtenerProveedoresActivos()
ctrl        --> b_ventana   : 3. CI_VentanaGestionarProveedorShow(listadoProveedores)

' -- Rama Buscar --
b_ventana   --> ctrl        : 4a. [BUSCAR] buscarProveedores(textoBusqueda)
ctrl        --> e_proveedor : 5a. obtenerProveedoresFiltrados(textoBusqueda)
ctrl        --> b_ventana   : 6a. CI_VentanaGestionarProveedorShow(listadoFiltrado)

' -- Rama Agregar (1º Cancelar, 2º GuardarNuevo) --
b_ventana   --> ctrl        : 4b. [AGREGAR] Envía btnNuevoProveedor
ctrl        --> b_form      : 5b. CI_FormularioProveedorShow() vacío
a_encargado --> b_form      : 6b-cancel. [alt-Cancelar] clickCancelar(btnCancelar) → cierra modal
b_form      --> ctrl        : 6b. [else-GuardarNuevo] ingresarDatos(..., btnGuardarNuevo)
ctrl        --> e_proveedor : 7b. verificarRucDuplicado(ruc)
ctrl        --> e_proveedor : 8b. registrarProveedor(razon_social, ruc, telefono, correo, direccion)
ctrl        --> e_proveedor : 9b. obtenerProveedoresActivos()
ctrl        --> b_ventana   : 10b. CI_VentanaGestionarProveedorShow(listadoProveedores)

' -- Rama Editar (1º Cancelar, 2º GuardarCambios) --
b_ventana   --> ctrl        : 4c. [EDITAR] Envía btnEditarProveedor
ctrl        --> e_proveedor : 5c. obtenerDatosProveedor(id_prov)
ctrl        --> b_form      : 6c. CI_FormularioProveedorShow(datosProveedor) con RUC sólo lectura
a_encargado --> b_form      : 7c-cancel. [alt-Cancelar] clickCancelar(btnCancelar) → cierra modal
b_form      --> ctrl        : 7c. [else-GuardarCambios] modificarDatos(..., btnGuardarCambios)
ctrl        --> e_proveedor : 8c. actualizarProveedor(id_prov, razon_social, telefono, correo, direccion)
ctrl        --> e_proveedor : 9c. obtenerProveedoresActivos()
ctrl        --> b_ventana   : 10c. CI_VentanaGestionarProveedorShow(listadoProveedores)

' -- Rama Eliminar (1º Cancelar, 2º Aceptar) --
b_ventana   --> ctrl        : 4d. [ELIMINAR] Envía btnEliminarProveedor
ctrl        --> b_confirm   : 5d. CI_MensajeConfirmacionShow()
a_encargado --> b_confirm   : 6d-cancel. [alt-Cancelar] clickCancelar(btnCancelar) → cierra confirmación
b_confirm   --> ctrl        : 6d. [else-Aceptar] Envía btnAceptar
ctrl        --> e_proveedor : 7d. verificarExistenciaActiva(id_prov)
ctrl        --> e_proveedor : 8d. desactivarProveedor(id_prov)
ctrl        --> e_proveedor : 9d. obtenerProveedoresActivos()
ctrl        --> b_ventana   : 10d. CI_VentanaGestionarProveedorShow(listadoProveedores)

' -- Mensajes de sistema --
ctrl        --> b_msg       : * CI_MensajeSistemaShow() en éxito o validación fallida
