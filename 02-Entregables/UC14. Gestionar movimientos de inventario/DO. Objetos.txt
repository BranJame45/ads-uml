ctrl        --> e_producto   : 8c. descontarStockFisico(id_prod, cantidad) / sumarStockFisico(id_prod, cantidad)
ctrl        --> e_movimiento : 9c. anularMovimiento(id_movimiento, id_usuario, motivo_anulacion)

ctrl        --> b_ventana    : 11c. CI_VentanaMovimientosShow(listadoMovimientos)
