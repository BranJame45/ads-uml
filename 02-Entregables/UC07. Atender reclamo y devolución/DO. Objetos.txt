' ============================================================
' RELACIONES NUMERADAS (orden del DS, priorizando camino negativo)
' ============================================================

' -- Flujo inicial (Directo Actor -> Controlador) --
a_vendedor --> ctrl      : 1. validarBtnAtenderReclamo(btnAtenderReclamo)
ctrl       --> b_ventana : 2. CI_VentanaAtenderReclamoShow()

' -- Rama Buscar Venta --
b_ventana  --> ctrl      : 3a. [BUSCAR] buscarVenta(num_comprobante)
ctrl       --> e_venta   : 4a. obtenerVentaPorComprobante(num_comprobante)
ctrl       --> e_detalle : 5a. obtenerDetalleVenta(id_venta)
ctrl       --> b_ventana : 6a. CI_VentanaAtenderReclamoShow(datosVenta, detalleVenta)

' -- Rama Registrar Reclamo (1º Cancelar, 2º ConfirmarReclamo) --
b_ventana  --> ctrl      : 3b. [RECLAMO] Envía btnRegistrarReclamo
ctrl       --> b_form    : 4b. CI_FormularioReclamoShow(datosVenta, detalleVenta)
a_vendedor --> b_form    : 5b-cancel. [alt-Cancelar] clickCancelar(btnCancelar) → cierra modal
b_form     --> ctrl      : 5b. [else-Confirmar] ingresarDatos(..., btnConfirmarReclamo)
ctrl       --> e_reclamo : 6b. registrarReclamo(id_venta, id_prod, id_usuario, motivo, cantidad, tipo_resolucion, monto_devuelto)
ctrl       --> e_producto: 7b. sumarStockFisico(id_prod, cantidad)
ctrl       --> e_producto: 8b. [opt Cambio de producto] descontarStockFisico(id_prod, cantidad)
ctrl       --> b_ventana : 9b. CI_VentanaAtenderReclamoShow()

' -- Mensajes de sistema --
ctrl       --> b_msg     : * CI_MensajeSistemaShow() en éxito, aviso o validación fallida
